set(CMAKE_CXX_STANDARD 20)

set(DEPENDENCY_ROOT "D:/works/golf_launch_monitor/thirdparty" CACHE PATH "Existing third-party source root")

  cmake_policy(SET CMP0135 NEW)

  URL_HASH SHA256=eb33e51f49a15e023950cd7825ca74a4a2b43db8354825ac24fc1b7ee09e6fa3
  SOURCE_SUBDIR build/cmake)
FetchContent_MakeAvailable(zstd)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)

add_library(glad STATIC "${DEPENDENCY_ROOT}/glad/generated/src/gl.c")
target_include_directories(glad PUBLIC "${DEPENDENCY_ROOT}/glad/generated/include")
add_library(imgui STATIC
    "${DEPENDENCY_ROOT}/imgui/imgui.cpp" "${DEPENDENCY_ROOT}/imgui/imgui_draw.cpp"
    "${DEPENDENCY_ROOT}/imgui/imgui_tables.cpp" "${DEPENDENCY_ROOT}/imgui/imgui_widgets.cpp"
    "${DEPENDENCY_ROOT}/imgui/backends/imgui_impl_glfw.cpp"
    "${DEPENDENCY_ROOT}/imgui/backends/imgui_impl_opengl3.cpp")
target_include_directories(imgui PUBLIC "${DEPENDENCY_ROOT}/imgui" "${DEPENDENCY_ROOT}/imgui/backends")
target_link_libraries(imgui PUBLIC glfw glad)
if(MSVC)
  target_compile_options(glfw PRIVATE /utf-8)
  target_compile_options(imgui PRIVATE /utf-8)
endif()
add_library(calibration_core src/calibration.cpp src/settings.cpp src/auto_capture.cpp src/ui_layout.cpp src/brightness.cpp src/stereo.cpp src/rig.cpp src/world_origin.cpp src/scene3d.cpp src/ball_detection.cpp src/recording.cpp)
target_include_directories(calibration_core PUBLIC src ${OpenCV_INCLUDE_DIRS} "${DEPENDENCY_ROOT}/nlohmann/include")
target_link_libraries(calibration_core PUBLIC ${OpenCV_LIBS} libzstd_static)
add_executable(boccia_calibration WIN32 src/main.cpp src/camera.cpp src/scene_renderer.cpp)
target_include_directories(boccia_calibration PRIVATE "${IMV_ROOT}/Development/Include/IMV")
target_link_libraries(boccia_calibration PRIVATE calibration_core imgui "${IMV_ROOT}/Development/Lib/x64/MVSDKmd.lib" opengl32 comdlg32 ole32 shell32)
target_compile_definitions(boccia_calibration PRIVATE GLFW_INCLUDE_NONE NOMINMAX BOCCIA_DATA_DIR="${CMAKE_SOURCE_DIR}/data")
if(MSVC)
  target_link_options(boccia_calibration PRIVATE /ENTRY:mainCRTStartup)
  target_compile_options(calibration_core PRIVATE /utf-8 /W4)
  target_compile_options(boccia_calibration PRIVATE /utf-8 /W4)
endif()
